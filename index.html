    <script>

        // 전역 변수 선언

        let map; 

        let placeSearchService; 

        let currentMarkers = [];       // 현재 지도에 표시된 마커 배열

        let infoWindow;               // 정보창 팝업 객체

        let currentCategory = '스터디카페'; // 기본 선택 카테고리 기본값


        // 페이지 로드 완료 시 초기화

        window.addEventListener('DOMContentLoaded', function() {

            if (typeof kakao !== 'undefined' && kakao.maps) {

                initMap();

            } else {

                alert("카카오 지도 라이브러리 로딩에 실패했습니다.");

            }

        });


        // 지도 생성 함수

        function initMap() {

            const mapContainer = document.getElementById('map'); 

            

            if (navigator.geolocation) {

                navigator.geolocation.getCurrentPosition(function(position) {

                    const lat = position.coords.latitude;

                    const lon = position.coords.longitude;

                    const userPosition = new kakao.maps.LatLng(lat, lon);

                    

                    const mapOptions = { center: userPosition, level: 5 };

                    map = new kakao.maps.Map(mapContainer, mapOptions);

                    setupServices();

                }, function() {

                    createMapWithDefault();

                });

            } else {

                createMapWithDefault();

            }

        }


        function createMapWithDefault() {

            const mapContainer = document.getElementById('map'); 

            const defaultPosition = new kakao.maps.LatLng(37.566826, 126.9786567); 

            const mapOptions = { center: defaultPosition, level: 5 };

            map = new kakao.maps.Map(mapContainer, mapOptions);

            setupServices();

        }


        function setupServices() {

            placeSearchService = new kakao.maps.services.Places();

            infoWindow = new kakao.maps.InfoWindow({ zIndex: 1 });

            

            // 처음 페이지가 열릴 때 기본값('명일동 스터디카페')으로 자동 검색 수행

            searchPlaces();

        }


        // 검색 버튼을 누르거나 엔터를 쳤을 때 실행되는 함수

        function handleSearch() {

            // 새롭게 검색어를 치면 첫 번째 탭(스터디카페)으로 UI 초기화 설정

            const tabs = document.querySelectorAll('.tab-item');

            tabs.forEach(tab => tab.classList.remove('active'));

            tabs[0].classList.add('active');

            currentCategory = '스터디카페';


            searchPlaces();

        }


        // 실제 카카오 API로 검색을 요청하는 함수

        function searchPlaces() {

            const areaKeyword = document.getElementById('keyword').value;


            if (!areaKeyword.trim()) {

                alert('동네 이름을 입력해 주세요!');

                return false;

            }


            // [동네 이름] + [선택된 탭 카테고리] 조합 (예: "명일동 + 식당")

            const finalSearchQuery = `${areaKeyword} ${currentCategory}`;


            placeSearchService.keywordSearch(finalSearchQuery, searchCallback);

        }


        // 탭 메뉴를 클릭했을 때 호출되는 함수

        function changeTab(categoryName, element) {

            // 모든 탭에서 active 스타일 클래스를 지우고 클릭한 탭에만 부여

            const tabs = document.querySelectorAll('.tab-item');

            tabs.forEach(tab => tab.classList.remove('active'));

            element.classList.add('active');


            // 현재 선택된 카테고리를 바꾸고 재검색

            currentCategory = categoryName;

            searchPlaces();

        }


        // 검색 완료 콜백 함수

        function searchCallback(data, status, pagination) {

            if (status === kakao.maps.services.Status.OK) {

                displayResults(data);

            } else if (status === kakao.maps.services.Status.ZERO_RESULT) {

                clearPreviousResults();

                document.getElementById('placesList').innerHTML = `<div style="padding:20px; color:#999; text-align:center;">'${currentCategory}' 결과가 없습니다.</div>`;

            } else {

                alert('검색 중 오류가 발생했습니다.');

            }

        }


        // 리스트와 지도 마커를 다시 그려주는 핵심 함수

        function displayResults(places) {

            const listContainer = document.getElementById('placesList');

            

            // 기존 결과와 마커를 지우기

            clearPreviousResults();

            listContainer.innerHTML = '';


            const mapBounds = new kakao.maps.LatLngBounds();


            for (let i = 0; i < places.length; i++) {

                const placePosition = new kakao.maps.LatLng(places[i].y, places[i].x);

                

                // 별 모양 커스텀 마커 이미지 적용

                const markerImageSrc = 'https://t1.daumcdn.net/localimg/localimages/07/2012/img/marker_normal.png'; 

                const markerImageSize = new kakao.maps.Size(24, 35);

                const customMarkerImage = new kakao.maps.MarkerImage(markerImageSrc, markerImageSize);


                const marker = new kakao.maps.Marker({

                    position: placePosition,

                    image: customMarkerImage,

                    map: map

                });


                currentMarkers.push(marker);


                // 리스트 아이템 HTML 추가

                const itemElement = document.createElement('div');

                itemElement.className = 'place-item';

                const phoneNumber = places[i].phone ? places[i].phone : '전화번호 없음';


                itemElement.innerHTML = `

                    <div class="place-name">${places[i].place_name}</div>

                    <div class="place-phone">📞 ${phoneNumber}</div>

                `;


                // 팝업 내용 정의

                const popupHtmlContent = `

                    <div class="popup-card">

                        <strong>${places[i].place_name}</strong><br>

                        <span style="font-size:11px; color:#666;">${places[i].address_name}</span><br>

                        <a href="${places[i].place_url}" target="_blank">상세정보 보기 &rarr;</a>

                    </div>

                `;


                // 이벤트 바인딩

                kakao.maps.event.addListener(marker, 'click', function() {

                    openPopup(marker, popupHtmlContent, placePosition);

                });


                itemElement.onclick = function() {

                    openPopup(marker, popupHtmlContent, placePosition);

                };


                listContainer.appendChild(itemElement);

                mapBounds.extend(placePosition);

            }


            map.setBounds(mapBounds);

        }


        // 팝업 생성 및 화면 이동

        function openPopup(marker, content, position) {

            infoWindow.setContent(content);

            infoWindow.open(map, marker);

            map.panTo(position);

        }


        // 기존 데이터 청소 함수

        function clearPreviousResults() {

            for (let i = 0; i < currentMarkers.length; i++) {

                currentMarkers[i].setMap(null);

            }

            currentMarkers = [];

            infoWindow.close();

        }

    </script>
